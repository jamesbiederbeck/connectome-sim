# Optional, GPU-only backend (doom/gpu.py). Not required by the default CPU
# pipeline in doom/requirements.txt and not installed by it.
#
# cupy-cuda12x wheels do not bundle the CUDA runtime shared libraries for
# every function cupy needs (notably cuSPARSE, used for synaptic delivery);
# those come from the separate nvidia-*-cu12 wheels below. At runtime, both
# CUDA_PATH and LD_LIBRARY_PATH must point at the installed nvidia wheels,
# e.g. (from the repo root, venv named .venv-neural):
#   V=.venv-neural/lib/python3.11/site-packages/nvidia
#   export CUDA_PATH="$(pwd)/$V/cuda_runtime"
#   export LD_LIBRARY_PATH="$(pwd)/$V/cublas/lib:$(pwd)/$V/cusolver/lib:$(pwd)/$V/cusparse/lib:$(pwd)/$V/cuda_nvrtc/lib:$(pwd)/$V/curand/lib:$(pwd)/$V/nvjitlink/lib"
# Tested against numpy==1.24.4 (the pin in doom/requirements.txt) and driver
# 580.95.05 on a GTX 1060 (Pascal, compute capability 6.1). A newer
# cupy-cuda12x (14.x) requires numpy>=2.3 and will conflict with the rest of
# this project's pinned CPU stack; do not upgrade it without re-validating
# numba/brian2 against the newer numpy.
cupy-cuda12x==13.3.0
nvidia-cublas-cu12==12.9.2.10
nvidia-cuda-nvrtc-cu12==12.9.86
nvidia-cuda-runtime-cu12==12.9.79
nvidia-curand-cu12==10.3.10.19
nvidia-cusolver-cu12==11.7.5.82
nvidia-cusparse-cu12==12.5.10.65
nvidia-nvjitlink-cu12==12.9.86
